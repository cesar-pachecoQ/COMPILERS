/* ==== SECCIÓN 1: CASOS VÁLIDOS ==== */

rutina principal {                      // palabra reservada + identificador
    // Declaración y asignación
    contador = 123;                    // literal entera
    precio = 45.67;                    // literal flotante
    mensaje = "Hola\nMundo";           // literal de cadena con escape
    miArreglo[10] = contador;          // acceso a arreglo y puntuación

    /* Bucle válido */
    mientras (contador < 130) haz {
        contador = contador + 1;
        resultado = contador * 2 / 4;
    }

    // Condicional válido
    si (contador >= 130) entonces {
        regresa contador;
    } otro {
        regresa 0;
    }
}
